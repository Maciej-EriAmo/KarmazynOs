# === name     new session (fresh store)
# >>> expr     input
# next line    exact expected output
# ! needle     expected: line starts with "blad:" AND contains needle

=== lambda-apply

=== closure
>>> (define add (lambda (a) (lambda (b) (+ a b))))
<closure (a)>
>>> (define add10 (add 10))

# Church 2, 3; plus = λm.λn.λf.λx. m f (n f x)
>>> (define ch-two (lambda (f) (lambda (x) (f (f x)))))
